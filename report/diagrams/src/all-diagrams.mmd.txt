%%{init: {"themeVariables": {"fontSize": "16px"}}}%%
classDiagram
  direction LR
  class TransactionRecord {
    +transactionId: string
    +timestamp: string
    +amount: number
    +type: string
    +channel?: string
    +accountId?, deviceId?, location?
  }
  class Transaction {
    +riskScore: number
    +risk: RiskLevel
    +status: TransactionStatus
  }
  class AnalysisResult {
    +riskScore: number
    +risk: RiskLevel
    +decision: Decision
    +factors: string[]
    +method: string
    +limitations: string[]
  }
  class TransactionFeatures {
    +amountLog1p: number
    +hourUtc: number
    +transactionsLast24Hours
    +locationChanged
  }
  class BehavioralBaseline {
    +historicalTransactionCount
    +averageAmount
    +mostCommonChannel
  }
  class TransactionRepository {
    <<interface>>
    +list() Transaction[]
    +save(t) void
    +clear() void
  }
  class LocalStorageRepository {
    -key: layernet-transactions
  }
  class TransactionService {
    -repository
    +list()
    +save(t)
    +analyze(record) AnalysisResult
  }
  class fraudAnalysis {
    <<module>>
    +analyzeTransaction(tx)
  }
  class featureEngineering {
    <<module>>
    +engineerTransactionFeatures()
    +createBehavioralBaseline()
  }
  TransactionRecord <|-- Transaction
  TransactionRepository <|.. LocalStorageRepository
  TransactionService o-- TransactionRepository
  TransactionService ..> fraudAnalysis : uses
  TransactionService ..> featureEngineering : uses
  TransactionService ..> AnalysisResult : creates
  AnalysisResult *-- TransactionFeatures
  AnalysisResult *-- BehavioralBaseline
  TransactionRepository ..> Transaction : stores

%%%FILE fig4_12_state
stateDiagram-v2
  [*] --> Idle
  Idle --> Editing : analyst types
  Editing --> Invalid : submit with errors
  Invalid --> Editing : correct fields
  Editing --> Analysing : submit valid (loading = true)
  Analysing --> Saved : analyze() and save() resolve
  Analysing --> Failed : exception (submitError set)
  Saved --> Displayed : result and saveMessage set
  Failed --> Editing : retry
  Displayed --> Idle : Reset
  Displayed --> Editing : edit and resubmit

%%%FILE fig4_19_seq_chat
sequenceDiagram
  actor Analyst
  participant Panel as AIChat panel
  participant Route as /api/chat
  participant Gemini as Gemini API
  Analyst->>Panel: type question, send
  Panel->>Route: POST messages (useChat transport)
  alt API key not configured
    Route-->>Panel: 503 "Gemini API key is not configured."
    Panel-->>Analyst: error message
  else key configured
    Route->>Gemini: streamText(system prompt, messages)
    loop streamed parts
      Gemini-->>Route: text chunk
      Route-->>Panel: UI message stream
      Panel-->>Analyst: incremental answer
    end
  end
  opt stop or regenerate
    Analyst->>Panel: Stop / Retry
  end
